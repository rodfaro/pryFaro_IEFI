CREATE DATABASE IEFI;
GO

USE IEFI;

CREATE TABLE Usuarios(
	idUsuario INT PRIMARY KEY IDENTITY(1,1),
	Usuario VARCHAR(25) NOT NULL,
	Passw VARCHAR(25) NOT NULL,
	Admin BIT NOT NULL,
	Nombre VARCHAR(50),
	Direccion VARCHAR(150),
	Celular VARCHAR(16)
);
GO

CREATE TABLE Auditar(
	Id INT PRIMARY KEY IDENTITY(1,1),
	IdUsuario INT,
	Fecha DATETIME DEFAULT GETDATE(),
	TiempoTrabajado INT NOT NULL,
	Descripcion VARCHAR(200),

	FOREIGN KEY (IdUsuario) REFERENCES Usuarios(idUsuario)
);
GO

CREATE TABLE Tareas(
	IdTarea INT PRIMARY KEY IDENTITY(1,1),
	IdUsuario INT,
	Tarea VARCHAR(50) NOT NULL,
	Fecha DATE DEFAULT CAST(GETDATE() AS DATE),
	Lugar VARCHAR(30),


	--Uniforme
	UniformeInsumo BIT,

	--Licencias
	LicenciaEstudio BIT,
	LicenciaVacacion BIT,
	LicenciaMedica BIT,
	LicenciaPersonal BIT,
	--Reclamo
	ReclamoSalario BIT,
	ReclamoRecibo BIT,
	--Otros
	HsExtra BIT,
	Ausencia BIT,
	PermisoTemporal BIT,
	Retraso BIT,

	Comentario VARCHAR(400),
	FOREIGN KEY (IdUsuario) REFERENCES Usuarios(idUsuario)
);
GO
